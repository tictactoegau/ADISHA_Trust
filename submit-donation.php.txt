<?php
/*
  ADISHA Trust Foundation Donation Form Backend
  ------------------------------------------------
  Purpose:
  - Receives donor information from donate.html
  - Saves each submission into donations.csv
  - Does NOT process payments yet
  - Can later be connected to Stripe, PayPal, Razorpay, Donorbox, etc.

  Required:
  - PHP hosting/server
  - Folder must be writable so donations.csv can be created
*/

declare(strict_types=1);

// Only allow POST requests
if ($_SERVER["REQUEST_METHOD"] !== "POST") {
    http_response_code(405);
    exit("Method Not Allowed");
}

// Basic helper to clean text input
function clean_input(string $value): string {
    $value = trim($value);
    $value = strip_tags($value);
    $value = htmlspecialchars($value, ENT_QUOTES, "UTF-8");
    return $value;
}

// Collect form values
$frequency      = clean_input($_POST["frequency"] ?? "");
$amount         = clean_input($_POST["amount"] ?? "");
$customAmount   = clean_input($_POST["custom_amount"] ?? "");
$program        = clean_input($_POST["program"] ?? "");
$firstName      = clean_input($_POST["first_name"] ?? "");
$lastName       = clean_input($_POST["last_name"] ?? "");
$email          = clean_input($_POST["email"] ?? "");
$phone          = clean_input($_POST["phone"] ?? "");
$message        = clean_input($_POST["message"] ?? "");
$anonymous      = isset($_POST["anonymous"]) ? "Yes" : "No";
$updates        = isset($_POST["updates"]) ? "Yes" : "No";

// If custom amount is entered, use that instead of selected amount
$finalAmount = $customAmount !== "" ? $customAmount : $amount;

// Basic validation
$errors = [];

if ($frequency === "") {
    $errors[] = "Contribution type is required.";
}

if ($finalAmount === "" || !is_numeric($finalAmount) || (float)$finalAmount <= 0) {
    $errors[] = "A valid contribution amount is required.";
}

if ($program === "") {
    $errors[] = "Program selection is required.";
}

if ($firstName === "") {
    $errors[] = "First name is required.";
}

if ($lastName === "") {
    $errors[] = "Last name is required.";
}

if ($email === "" || !filter_var($email, FILTER_VALIDATE_EMAIL)) {
    $errors[] = "A valid email address is required.";
}

// If errors exist, show simple response
if (!empty($errors)) {
    http_response_code(400);
    echo "<h1>Donation Form Error</h1>";
    echo "<p>Please correct the following:</p>";
    echo "<ul>";

    foreach ($errors as $error) {
        echo "<li>" . htmlspecialchars($error, ENT_QUOTES, "UTF-8") . "</li>";
    }

    echo "</ul>";
    echo '<p><a href="donate.html">Go back to donation page</a></p>';
    exit;
}

// Prepare CSV file
$csvFile = __DIR__ . "/donations.csv";
$fileExists = file_exists($csvFile);

// Submission metadata
$submittedAt = date("Y-m-d H:i:s");
$ipAddress = $_SERVER["REMOTE_ADDR"] ?? "Unknown";

// CSV row
$row = [
    $submittedAt,
    $frequency,
    $finalAmount,
    $program,
    $firstName,
    $lastName,
    $email,
    $phone,
    $anonymous,
    $updates,
    $message,
    $ipAddress
];

// Open CSV file
$file = fopen($csvFile, "a");

if ($file === false) {
    http_response_code(500);
    exit("Unable to save donation information. Please contact the site administrator.");
}

// Add header row if file is new
if (!$fileExists) {
    fputcsv($file, [
        "Submitted At",
        "Frequency",
        "Amount",
        "Program",
        "First Name",
        "Last Name",
        "Email",
        "Phone",
        "Anonymous",
        "Receive Updates",
        "Message",
        "IP Address"
    ]);
}

// Save donor submission
fputcsv($file, $row);
fclose($file);

header("Location: thank-you.html");
exit;
?>
